function changeRole(newRole) {
  if (newRole !== 'viewer') {
    const pwd = prompt(`Mot de passe pour le profil ${newRole.toUpperCase()} :`);
    if (pwd !== PASSWORDS[newRole]) {
      alert("Mot de passe incorrect.");
      document.getElementById('role-selector').value = currentRole;
      return;
    }
  }
  currentRole = newRole;
  localStorage.setItem('app_role', currentRole);
  document.getElementById('role-selector').value = currentRole;
  applyRolePermissions();
}

function applyRolePermissions() {

// --- NETTOYAGE AUTOMATIQUE DES DOUBLONS ---
window.cleanDuplicates = function() {
  if (currentRole === 'viewer' || currentRole === 'cfo') return;
  
  if (!confirm("Voulez-vous rechercher et supprimer automatiquement tous les doublons de la base de données ?\n\n(L'ouvrier assigné à une équipe de chantier sera conservé en priorité)")) return;
  
  let seenIds = new Set();
  let seenNames = new Set();
  let removedCount = 0;

  const isDup = (w) => {
      const normId = String(w.id || '').trim().toUpperCase();
      const normName = String(w.name || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim().toUpperCase().replace(/\s+/g, ' ');
      
      if ((normId !== '' && seenIds.has(normId)) || (normName !== '' && seenNames.has(normName))) {
          return true; 
      }
      
      if (normId !== '') seenIds.add(normId);
      if (normName !== '') seenNames.add(normName);
      return false;
  };

  cardsData.forEach(c => {
      if (c.tasks) {
          c.tasks.forEach(t => {
              if (t.workerList) {
                  t.workerList = t.workerList.filter(w => {
                      if (isDup(w)) { removedCount++; return false; }
                      return true;
                  });
              }
          });
      }
  });

  unassignedWorkers = unassignedWorkers.filter(w => {
      if (isDup(w)) { removedCount++; return false; }
      return true;
  });

  if (removedCount > 0) {
      alert(`✅ Nettoyage terminé : ${removedCount} doublon(s) supprimé(s).`);
      recalculateTotals();
      renderCards();
      renderDrawer();
      if(document.getElementById('db-modal').style.display === 'flex') renderDBTable();
      syncToFirebase();
  } else {
      alert("👌 Aucun doublon n'a été trouvé dans la base de données !");
  }
};

function filterDBTable(query) {
  const lowerQuery = query.toLowerCase();
  const rows = document.querySelectorAll('#db-table-body tr');
  
  rows.forEach(row => {
     const idText = row.cells[0].querySelector('input').value.toLowerCase();
     const nameText = row.cells[1].querySelector('input').value.toLowerCase();
     const companyText = row.cells[2].querySelector('select').value.toLowerCase();
     
     if (idText.includes(lowerQuery) || nameText.includes(lowerQuery) || companyText.includes(lowerQuery)) {
        row.style.display = '';
     } else {
        row.style.display = 'none';
     }
  });
}

function renderDBTable() {
  const tbody = document.getElementById('db-table-body');
  let html = '';
  
  const addRow = (w, context, cIdx, tIdx, supName) => {

     html += `<tr>

           <button class="btn btn-outline" style="color:#dc2626; border-color:#fca5a5; padding:2px 8px;" onclick="deleteWorkerFromDB('${w.id}', '${context}', ${cIdx}, ${tIdx})">&times;</button>

     </tr>`;

  unassignedWorkers.forEach(w => addRow(w, 'unassigned', -1, -1, ''));
  cardsData.forEach((c, cIdx) => {
     c.tasks.forEach((t, tIdx) => {

function editWorker(oldId, field, newVal, context, cIdx, tIdx, inputEl) {
  let w = context === 'unassigned' ? unassignedWorkers.find(x => x.id === oldId) : cardsData[cIdx].tasks[tIdx].workerList.find(x => x.id === oldId);
  if (!w) return;

  let dup = { found: false };
  
  if (field === 'id') {
    dup = findWorkerDuplicate(newVal.trim(), null, oldId);
  } else if (field === 'name') {
    dup = findWorkerDuplicate(null, newVal.trim(), oldId);
  }

  if (dup.found) {
    const msg = `⚠️ DOUBLON DÉTECTÉ !\n\nL'information "${newVal}" est déjà utilisée ${dup.location} par l'ouvrier [${dup.worker.name}].\n\nConserver ces deux lignes va fausser les effectifs du chantier.\n\nVoulez-vous SUPPRIMER définitivement la ligne que vous êtes en train de modifier pour nettoyer la base ?`;
    
    if (confirm(msg)) {
       if (context === 'unassigned') {
         const index = unassignedWorkers.findIndex(x => x.id === oldId);
         if(index !== -1) unassignedWorkers.splice(index, 1);
       } else {
         const index = cardsData[cIdx].tasks[tIdx].workerList.findIndex(x => x.id === oldId);
         if(index !== -1) cardsData[cIdx].tasks[tIdx].workerList.splice(index, 1);
       }
       syncToFirebase();
       renderCards();
       renderDrawer();
       renderDBTable();
       return;
    } else {
       if (inputEl) inputEl.value = w[field];
       return;
    }
  }

  w[field] = newVal.trim();
  syncToFirebase();
  renderCards();
  renderDrawer();
}

function deleteWorkerFromDB(wId, context, cIdx, tIdx) {
  if(!confirm("Supprimer cet ouvrier DÉFINITIVEMENT de la base ?")) return;
  if (context === 'unassigned') {
     const index = unassignedWorkers.findIndex(x => x.id === wId);
     if(index !== -1) unassignedWorkers.splice(index, 1);
  } else {
     const index = cardsData[cIdx].tasks[tIdx].workerList.findIndex(x => x.id === wId);
     if(index !== -1) cardsData[cIdx].tasks[tIdx].workerList.splice(index, 1);
  }
  syncToFirebase();
  renderCards();
  renderDrawer();
  renderDBTable();
}

function renderCards() {
  try {
    updateFilterDropdowns(); // Met à jour les menus dynamiques MGR et SUP
    
    const alphaGrid = document.getElementById('grid-alpha');
    if (alphaGrid) alphaGrid.innerHTML = "";

    CATEGORY_INFO.forEach(cat => {
      const grid = document.getElementById(`grid-${cat.id}`);
      if(grid) grid.innerHTML = "";
    });

    let displayList = cardsData.map((card, cIdx) => ({ card, cIdx }));

    // FILTRES GLOBAUX AVANT AFFICHAGE
    displayList = displayList.filter(item => {
        if(currentFilterCat !== 'all' && currentFilterCat !== 'alpha' && item.card.category !== currentFilterCat) return false;
        
        let mgrVal = item.card.manager ? item.card.manager.trim().toUpperCase() : '';
        if(currentFilterMgr !== 'ALL' && mgrVal !== currentFilterMgr) return false;
        
        let supVal = item.card.name ? item.card.name.trim() : '';
        if(currentFilterSup !== 'ALL' && supVal !== currentFilterSup) return false;
        
        return true;
    });

    if (currentFilterCat === 'alpha') {
      displayList.sort((a, b) => {
        let tsA = (a.card.timesheetNo || '').trim();
        let tsB = (b.card.timesheetNo || '').trim();

        if (tsA && tsB) {
          return tsA.localeCompare(tsB, undefined, { numeric: true, sensitivity: 'base' });
        }
        if (tsA && !tsB) return -1;
        if (!tsA && tsB) return 1;

        let nameA = (a.card.name || '').trim();
        let nameB = (b.card.name || '').trim();
        return nameA.localeCompare(nameB, undefined, { numeric: true, sensitivity: 'base' });
      });
    }

    displayList.forEach(({ card, cIdx }) => {
      if(!card.tasks) card.tasks = [];

      let tasksHtml = card.tasks.map((t, tIdx) => {

            else if(w.company === 'DSS') colorClass = 'dss';

      let cardLinkedSum = 0;

         const hasM = t.workers !== undefined && t.workers !== null && t.workers !== "";

function handleSupChange(cIdx, selectEl) {
    if (currentRole === 'viewer' || currentRole === 'cfo') {
        selectEl.value = cardsData[cIdx].name; 
        return;
    }
    const val = selectEl.value;
    if (val === '__NEW__') {
        const newName = prompt("Nom du nouveau Superviseur (ex: VASSILIS TSIRKAS) :");
        if (newName && newName.trim() !== '') {
            const upperName = newName.trim().toUpperCase();
            if (!siteSupervisors.includes(upperName)) {
                siteSupervisors.push(upperName);
                siteSupervisors.sort();
            }
            cardsData[cIdx].name = upperName;
        } else {
            selectEl.value = cardsData[cIdx].name; 
            return;
        }
    } else {
        cardsData[cIdx].name = val;
    }
    syncToFirebase();
    renderCards();
    if(currentView === 'orgchart') renderOrgChart();
}

function renderOrgChart() {
  const container = document.getElementById('orgchart-view');
  const groups = {};
  
  cardsData.forEach(c => {
     let mgr = (c.manager && c.manager.trim() !== '') ? c.manager.trim().toUpperCase() : 'SANS RESPONSABLE';
     if(!groups[mgr]) groups[mgr] = [];
     groups[mgr].push(c);
  });
  
  let html = `<div class="org-level-top"><span>CHANTIER RIVIERA TOWER</span></div><div class="org-managers-grid">`;
  
  Object.keys(groups).sort().forEach(mgr => {
     let mgrTotalWorkers = 0;
     let supsHtml = '';
     
     groups[mgr].forEach(sup => {
        let supWorkers = 0;
        let foremenGroups = {};
        let directWkrs = 0;

        sup.tasks.forEach(t => { 
          const linked = t.workerList ? t.workerList.length : 0;
          const hasM = t.workers !== undefined && t.workers !== null && t.workers !== "";
          const manual = hasM ? parseInt(t.workers) : linked;

        });
        if (directWkrs > 0) {

            : `<div style="font-size:12px; font-weight:700; color:#94a3b8; margin-bottom:4px; text-align:center;">TS: Non défini</div>`;

        supsHtml += `<div class="org-sup-card" style="border-left-color: ${catColor}">
            ${tsNameDisplay}
            <div class="org-sup-name" style="margin-bottom:2px; font-size:12px; color:#475569; text-align:center; text-transform:uppercase;">👷‍♂️ ${sup.name}</div>

        <div class="org-manager-name">${mgr === 'SANS RESPONSABLE' ? '⚠️ SANS RESPONSABLE' : '👔 ' + mgr}</div>

// --- GESTION DU MULTI-SELECT ET DRAG DROP (Modifié pour les Tâches) ---
function initWorkerSortables() {
  workerSortables.forEach(s => s.destroy());
  workerSortables = [];
  if (currentRole === 'viewer' || currentRole === 'cfo') return;

  // L'initialisation se fait maintenant par `.task-worker-dropzone`
  document.querySelectorAll('.task-worker-dropzone, #unassigned-dropzone').forEach(el => {
    const s = new Sortable(el, {
      group: 'workers-pool',
      animation: 150,
      multiDrag: true,
      selectedClass: 'selected-pill',
      fallbackTolerance: 3,
      forceFallback: true, // Améliore considérablement le scroll horizontal lointain
      scroll: true, 
      bubbleScroll: true,
      onSelect: function() { updateFab(); },
      onDeselect: function() { updateFab(); },
      onEnd: function(evt) { handleWorkerDrop(evt); }
    });
    workerSortables.push(s);
  });
}

function handleWorkerDrop(evt) {
  if (currentRole === 'viewer' || currentRole === 'cfo') return;
  if (evt.from === evt.to) {
     updateFab();
     return;
  }

  const movedItems = (evt.items && evt.items.length > 0) ? evt.items : [evt.item];
  const workersToMove = [];

  movedItems.forEach(item => {
    const wId = item.dataset.wid;
    const wName = item.dataset.wname;
    const wComp = item.dataset.wcomp;
    workersToMove.push({ id: wId, name: wName, company: wComp });
  });

  // Suppression des anciens emplacements
  if (evt.from.id === 'unassigned-dropzone') {
     unassignedWorkers = unassignedWorkers.filter(w => !workersToMove.some(m => m.id === w.id));
  } else {
     const fromCIdx = parseInt(evt.from.dataset.cidx);
     const oldTIdx = parseInt(evt.from.dataset.tidx);
     cardsData[fromCIdx].tasks[oldTIdx].workerList = cardsData[fromCIdx].tasks[oldTIdx].workerList.filter(w => !workersToMove.some(m => m.id === w.id));
  }

  // Ajout dans les nouveaux emplacements (par TÂCHE désormais)
  if (evt.to.id === 'unassigned-dropzone') {
     unassignedWorkers.push(...workersToMove);
  } else {
     const toCIdx = parseInt(evt.to.dataset.cidx);
     const toTIdx = parseInt(evt.to.dataset.tidx);
     if (!cardsData[toCIdx].tasks[toTIdx].workerList) cardsData[toCIdx].tasks[toTIdx].workerList = [];
     cardsData[toCIdx].tasks[toTIdx].workerList.push(...workersToMove);
  }

  recalculateTotals();
  renderCards();
  renderDrawer();
  updateFab();
  syncToFirebase();
}

function updateFab() {
  const selected = document.querySelectorAll('.worker-pill.sortable-selected, .worker-pill.selected-pill');
  const fabMove = document.getElementById('fab-move');
  const fabCount = document.getElementById('fab-count');
  
  if (selected.length > 0) {
    fabMove.style.display = 'flex';
    fabCount.innerText = selected.length;
  } else {
    fabMove.style.display = 'none';
  }
}

function confirmWorkerMove() {
  if (currentRole === 'viewer' || currentRole === 'cfo') return;
  const targetVal = document.getElementById('modal-sup-select').value;
  const selectedPills = document.querySelectorAll('.worker-pill.sortable-selected, .worker-pill.selected-pill');
  
  let targetCIdx = -1;
  let targetTIdx = -1;
  
  if(targetVal !== "-1") {
      let parts = targetVal.split('-');
      targetCIdx = parseInt(parts[0]);
      targetTIdx = parseInt(parts[1]);
  }
  
  selectedPills.forEach(pill => {
      const wId = pill.dataset.wid;
      const wName = pill.dataset.wname;
      const wComp = pill.dataset.wcomp;
      const cIdx = pill.dataset.cidx === "unassigned" ? -1 : parseInt(pill.dataset.cidx);
      const tIdx = pill.dataset.tidx === "unassigned" ? -1 : parseInt(pill.dataset.tidx);
      
      if (targetCIdx === cIdx && targetTIdx === tIdx) return; 
      
      const workerObj = { id: wId, name: wName, company: wComp };
      
      if (cIdx === -1) { 
         unassignedWorkers = unassignedWorkers.filter(w => w.id !== wId); 
      } else {
         if (cardsData[cIdx] && cardsData[cIdx].tasks[tIdx]) {
            cardsData[cIdx].tasks[tIdx].workerList = cardsData[cIdx].tasks[tIdx].workerList.filter(w => w.id !== wId);
         }
      }
      
      if (targetCIdx === -1) { 
         if(!unassignedWorkers.find(w => w.id === wId)) {
             unassignedWorkers.push(workerObj); 
         }
      } else {
         if (!cardsData[targetCIdx].tasks[targetTIdx].workerList) cardsData[targetCIdx].tasks[targetTIdx].workerList = [];
         cardsData[targetCIdx].tasks[targetTIdx].workerList.push(workerObj);
      }
  });
  
  closeWorkerModal(); 
  document.getElementById('fab-move').style.display = 'none';
  recalculateTotals(); 
  renderCards(); 
  renderDrawer(); 
  syncToFirebase();
}

  const dz = document.getElementById('unassigned-dropzone');
  if (!dz) return;
  
  dz.innerHTML = unassignedWorkers.map(w => {
    let colorClass = 'other';
    if(w.company === 'SKYBUILDING') colorClass = 'skybuilding';
    else if(w.company === 'DSS') colorClass = 'dss';
    else if(w.company === 'EUROASIA') colorClass = 'euroasia';
    
    return `<div class="worker-pill ${colorClass}" data-wid="${w.id}" data-wname="${w.name}" data-wcomp="${w.company}" data-cidx="unassigned" data-tidx="unassigned">
      ${w.id} - ${w.name}
    </div>`;
  }).join('');

  const fabDrawerCount = document.getElementById('fab-drawer-count');
  if (fabDrawerCount) fabDrawerCount.innerText = unassignedWorkers.length;
}

function importWorkersDB(event) {
  if (currentRole === 'viewer' || currentRole === 'cfo') return;
  const file = event.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const data = new Uint8Array(e.target.result);
      const workbook = XLSX.read(data, { type: 'array' });
      let newWorkersCount = 0;

      workbook.SheetNames.forEach(sheetName => {
        if (!sheetName) return;
        const worksheet = workbook.Sheets[sheetName];
        if (!worksheet) return;

        const rawRows = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: "" });
        if (!rawRows || rawRows.length === 0) return;

        let headerRowIdx = -1;
        for (let r = 0; r < rawRows.length; r++) {
          const rowArray = rawRows[r];
          if (!Array.isArray(rowArray)) continue;
          const lineStr = rowArray.map(cell => String(cell || "").toUpperCase()).join(" ");
          if (lineStr.includes("NAME") || lineStr.includes("SURNAME") || lineStr.includes("NOM")) { headerRowIdx = r; break; }
        }
        if (headerRowIdx === -1) return;

        const headerRow = rawRows[headerRowIdx] || [];
        let idCol = -1, companyCol = -1, workerCols = [];
        for (let c = 0; c < headerRow.length; c++) {
          const colName = String(headerRow[c] || "").trim().toUpperCase();
          if ((colName.includes("COMET") || colName === "NO." || colName === "ID" || colName.includes("NEOAXESS")) && idCol === -1) idCol = c;
          if (colName.includes("NAME") || colName.includes("SURNAME") || colName.includes("NOM")) workerCols.push(c);
          if (colName.includes("COMPANY") || colName.includes("SUBCONTRACTOR") || colName === "CO." || colName.includes("ENTREPRISE")) companyCol = c;
        }

        if (workerCols.length === 0) return;

        for (let r = headerRowIdx + 1; r < rawRows.length; r++) {
          const row = rawRows[r];
          if (!Array.isArray(row) || row.length === 0) continue;

          let workerName = "";
          workerCols.forEach(wc => { let val = String(row[wc] || "").trim(); if(val) workerName += val + " "; });
          workerName = workerName.trim();
          if (!workerName) continue;

          let workerId = idCol !== -1 ? String(row[idCol] || "").trim() : "";
          if (!workerId) workerId = "W-" + Math.floor(Math.random()*100000);

          let workerComp = "OTHER";
          if (companyCol !== -1 && String(row[companyCol] || "").trim() !== "") {
            workerComp = String(row[companyCol]).trim().toUpperCase();
            if (!siteCompanies.includes(workerComp)) siteCompanies.push(workerComp);
          }

          let isAssigned = false;
          cardsData.forEach(c => {
            if (c.tasks) {
              c.tasks.forEach(t => {
                if (t.workerList && t.workerList.find(w => w.id === workerId || w.name.toUpperCase() === workerName.toUpperCase())) isAssigned = true;
              });
            }
          });
          
          let isUnassigned = unassignedWorkers.find(w => w.id === workerId || w.name.toUpperCase() === workerName.toUpperCase());

          if (!isAssigned && !isUnassigned) {
            unassignedWorkers.push({ id: workerId, name: workerName, company: workerComp });
            newWorkersCount++;
          }
        }
      });

      renderCompanySelects();
      renderPayrollTable();
      renderDrawer(); 
      if(document.getElementById('db-modal').style.display === 'flex') renderDBTable();
      syncToFirebase();
      if (newWorkersCount > 0) alert(`Succès : ${newWorkersCount} nouveaux ouvriers ajoutés à la base !`);
      else alert("Aucun nouvel ouvrier ajouté (Ils sont peut-être déjà tous dans la base ou assignés).");

    } catch (err) { alert("Erreur technique : " + err.message); }
    event.target.value = "";
  };
  reader.readAsArrayBuffer(file);
}
