// Variables globales de base (ne pas écraser si elles sont déjà peuplées)
window.selectedTimesheets = window.selectedTimesheets || []; // Stocke les fiches cochées manuellement

// 2. CORRECTION ERREUR LIGNE 1 (handleCheck dans le scope global)
window.handleCheck = function(checkboxElement, cardId) {
    const id = cardId || checkboxElement.value;
    if (!id) return;

    if (checkboxElement.checked) {
        if (!window.selectedTimesheets.includes(id)) {
            window.selectedTimesheets.push(id);
        }
    } else {
        window.selectedTimesheets = window.selectedTimesheets.filter(item => item !== id);
    }
};

// 3. GESTION DE LA MULTI-SÉLECTION ET DES FILTRES
window.getMultiSelectValues = function(selectElementId) {
    const select = document.getElementById(selectElementId);
    if (!select) return [];
    // Transforme la HTMLCollection des options sélectionnées en tableau de valeurs
    return Array.from(select.selectedOptions).map(opt => opt.value);
};

window.applyFilters = function() {
    const selectedManagers = window.getMultiSelectValues('managerSelect');
    const selectedSupervisors = window.getMultiSelectValues('supervisorSelect');
    
    // Application des filtres dynamiques combinés sur l'objet cardsData[cite: 1].
    const filteredCards = window.cardsData.filter(card => {
        const matchManager = selectedManagers.length === 0 || selectedManagers.includes(card.manager);
        const matchSupervisor = selectedSupervisors.length === 0 || selectedSupervisors.includes(card.name);
        return matchManager && matchSupervisor;
    });

    // Rendu des cartes filtrées (ta fonction existante)
    if (typeof window.renderCards === 'function') {
        window.renderCards(filteredCards);
    }
};

// 4. GESTION DES IMPRESSIONS
window.printDisplayedTimesheets = function() {
    // Vérification stricte : le viewer et le cfo ne peuvent pas imprimer[cite: 1].
    if (window.currentRole !== 'timekeeper' && window.currentRole !== 'admin') {
        alert("Action refusée : Seuls le Timekeeper et l'Admin peuvent imprimer les fiches[cite: 1].");
        return;
    }
    
    let cardsToPrint = window.cardsData;
    
    // Filtre pour n'imprimer que les fiches cochées si l'utilisateur en a sélectionné
    if (window.selectedTimesheets && window.selectedTimesheets.length > 0) {
        cardsToPrint = window.cardsData.filter(card => window.selectedTimesheets.includes(card.name));
    }

    if (cardsToPrint.length === 0) {
        alert("Aucune fiche trouvée pour l'impression.");
        return;
    }

    // Appel de l'algorithme d'impression paginé défini dans l'architecture[cite: 1].
    if (typeof window.generateAndPrintTimesheets === 'function') {
        window.generateAndPrintTimesheets(cardsToPrint);
    }
};

// NOUVELLE FONCTION : Impression d'une seule fiche depuis un bouton spécifique
window.printSingleTimesheet = function(timesheetName) {
    // Vérification stricte des permissions[cite: 1].
    if (window.currentRole !== 'timekeeper' && window.currentRole !== 'admin') {
        alert("Action refusée : Seuls le Timekeeper et l'Admin peuvent imprimer[cite: 1].");
        return;
    }

    // Recherche de l'équipe cible dans l'Array d'équipes/superviseurs[cite: 1].
    const singleCard = window.cardsData.find(card => card.name === timesheetName);
    
    if (!singleCard) {
        console.error("Fiche introuvable :", timesheetName);
        return;
    }

    // Impression de cette unique fiche[cite: 1].
    if (typeof window.generateAndPrintTimesheets === 'function') {
        window.generateAndPrintTimesheets([singleCard]);
    }
};
