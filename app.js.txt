// --- 2. GESTION DES FILTRES ET MULTI-SÉLECTION ---
// Nouvelle fonction pour récupérer plusieurs valeurs depuis un <select multiple>
window.getMultiSelectValues = function(selectElementId) {
    const select = document.getElementById(selectElementId);
    if (!select) return [];
    return Array.from(select.selectedOptions).map(opt => opt.value);
};

// Application des filtres dynamiques combinés[cite: 1].
window.applyFilters = function() {
    const selectedManagers = window.getMultiSelectValues('managerSelect');
    const selectedSupervisors = window.getMultiSelectValues('supervisorSelect');
    const selectedCategory = document.getElementById('categorySelect') ? document.getElementById('categorySelect').value : '';
    
    const filteredCards = window.cardsData.filter(card => {
        const matchManager = selectedManagers.length === 0 || selectedManagers.includes(card.manager);
        const matchSupervisor = selectedSupervisors.length === 0 || selectedSupervisors.includes(card.name);
        const matchCategory = selectedCategory === '' || card.category === selectedCategory;
        
        return matchManager && matchSupervisor && matchCategory;
    });

    window.renderCards(filteredCards);
};


// --- 3. GESTION DE L'IMPRESSION DES TIMESHEETS ---
window.printDisplayedTimesheets = function() {
    // Vérification des droits : Seuls timekeeper et admin peuvent imprimer[cite: 1].
    if (window.currentRole !== 'timekeeper' && window.currentRole !== 'admin') {
        alert("Action refusée : Seuls le Timekeeper et l'Admin peuvent imprimer les Timesheets.");
        return;
    }
    
    // Impression de toutes les cartes affichées
    window.generateAndPrintTimesheets(window.cardsData);
};

// Nouvelle fonction pour imprimer une seule timesheet
window.printSingleTimesheet = function(timesheetName) {
    // Vérification des droits : Seuls timekeeper et admin peuvent imprimer[cite: 1].
    if (window.currentRole !== 'timekeeper' && window.currentRole !== 'admin') {
        alert("Action refusée : Seuls le Timekeeper et l'Admin peuvent imprimer les Timesheets.");
        return;
    }

    // Recherche de la carte spécifique
    const singleCard = window.cardsData.find(card => card.timesheetName === timesheetName || card.name === timesheetName);
    
    if (!singleCard) {
        console.error("Timesheet introuvable :", timesheetName);
        return;
    }

    // Impression de la carte unique
    window.generateAndPrintTimesheets([singleCard]);
};

// --- 4. RENDU DU DOM (GÉNÉRATION DES CARTES) ---
window.renderCards = function(cards) {
    const container = document.getElementById('cards-container');
    if (!container) return;
    
    container.innerHTML = '';
    
    cards.forEach(card => {
        const cardEl = document.createElement('div');
        cardEl.className = 'card';
        cardEl.dataset.category = card.category;
        
        let html = `<h3>${card.name} (Manager: ${card.manager})</h3>`;
        html += `<p>Catégorie : ${card.category}</p>`;
        
        // Affichage du bouton d'impression individuelle uniquement pour timekeeper et admin[cite: 1].
        if (window.currentRole === 'timekeeper' || window.currentRole === 'admin') {
            html += `<button class="print-single-btn" onclick="window.printSingleTimesheet('${card.name}')">🖨️ Imprimer la fiche</button>`;
        }

        html += `<div class="tasks-container">`;
        
        // Affichage des sous-équipes ou tâches[cite: 1].
        if (card.tasks && Array.isArray(card.tasks)) {
            card.tasks.forEach(task => {
                html += `<div class="task">
                            <h4>${task.foreman} - ${task.desc} (${task.source})</h4>
                            <p>Effectif cible : ${task.workers}</p>
                            <ul class="worker-list">`;
                
                // Affichage des ouvriers réellement assignés[cite: 1].
                if (task.workerList && Array.isArray(task.workerList)) {
                    task.workerList.forEach(worker => {
                        html += `<li>${worker.name} (${worker.company})</li>`;
                    });
                }
                
                html += `   </ul>
                         </div>`;
            });
        }
        html += `</div>`;
        
        cardEl.innerHTML = html;
        container.appendChild(cardEl);
    });
    
    window.initDragAndDrop();
};


// --- 5. DRAG & DROP ---
window.initDragAndDrop = function() {
    // Le viewer et le cfo ne peuvent pas réorganiser le chantier[cite: 1].
    if (window.currentRole === 'viewer' || window.currentRole === 'cfo') return;

    const lists = document.querySelectorAll('.worker-list');
    lists.forEach(list => {
        if (typeof Sortable !== 'undefined') {
            // SortableJS avec MultiDrag activé pour le Drag & Drop[cite: 1].
            new Sortable(list, {
                group: 'shared',
                multiDrag: true,
                selectedClass: 'selected',
                animation: 150,
                onEnd: function(evt) {
                    console.log("Élément déplacé. Mise à jour Firestore requise.");
                }
            });
        }
    });
};
